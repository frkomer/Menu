Main Menu (Ana Menuye Don)" -ForegroundColor DarkGray Write-Host "" $altSecim = Read-Host "Select Option (0-3 veya 99)" if ($altSecim -eq "0") { break } elseif ($altSecim -eq "1") { Write-Host "" $cs = Get-CimInstance Win32_ComputerSystem $bios = Get-CimInstance Win32_BIOS $cpu = Get-CimInstance Win32_Processor $os = Get-CimInstance Win32_OperatingSystem $ramGB = [math]::Round($cs.TotalPhysicalMemory / 1GB, 1) $oemKey = (Get-CimInstance -Query 'select * from SoftwareLicensingService').OA3xOriginalProductKey Write-Host "Marka / Model : $($cs.Manufacturer) - $($cs.Model)" -ForegroundColor Yellow Write-Host "Seri No : $($bios.SerialNumber)" -ForegroundColor Yellow Write-Host "Islemci / RAM : $($cpu.Name) / $ramGB GB" -ForegroundColor Yellow Write-Host "Windows : $($os.Caption) ($($os.OSArchitecture))" -ForegroundColor Yellow if ($oemKey) { Write-Host "BIOS OEM Key : $oemKey" -ForegroundColor Green } cscript //nologo C:\Windows\System32\slmgr.vbs /xpr Read-Host "Geri donmek icin Enter, Ana Menu icin 99" | Out-Null } elseif ($altSecim -eq "2") { Write-Host "" Get-PhysicalDisk | Select-Object FriendlyName, MediaType, HealthStatus, @{Name="Size(GB)";Expression={[math]::Round($_.Size/1GB,1)}} | Format-Table -AutoSize $batYol = "C:\battery-report.html" powercfg /batteryreport /output $batYol | Out-Null if (Test-Path $batYol) { Start-Process $batYol } Read-Host "Geri donmek icin Enter, Ana Menu icin 99" | Out-Null } elseif ($altSecim -eq "3") { Write-Host "" Get-NetIPAddress -AddressFamily IPv4 | Where-Object { $_.InterfaceAlias -notmatch "Loopback" } | Select-Object InterfaceAlias, IPAddress | Format-Table -AutoSize $paylasim = Read-Host "Dosya-yazici paylasimi acilsin mi? (E / H / 0: Geri / 99: Ana Menu)" if ($paylasim -eq "0") { continue } elseif ($paylasim -in @("E", "e", "Y", "y")) { netsh advfirewall firewall set rule group="Network Discovery" new enable=Yes | Out-Null netsh advfirewall firewall set rule group="File and Printer Sharing" new enable=Yes | Out-Null Write-Host "Paylasim aktif edildi!" -ForegroundColor Green } Read-Host "Geri donmek icin Enter, Ana Menu icin 99" | Out-Null } } } $script:altAracAktif = $false function Read-Host { param([string]$Prompt) if ($Prompt) { Write-Host -NoNewline "$Prompt`: " } $girdi = [Console]::ReadLine() if ($girdi -eq "99") { throw "ANA_MENU_DON" } if ($script:altAracAktif -and $girdi -eq "0") { throw "ALT_MENU_DON" } return $girdi } function Araci-Calistir ($depoAdi) { $p1 = "ht" + "tps://" $p2 = "frk" + "omer" $p3 = ".git" + "hub.io/" $tamUrl = $p1 + $p2 + $p3 + $depoAdi + "/index.html" while ($true) { $kod = "" if (Get-Command curl.exe -ErrorAction SilentlyContinue) { $kod = (curl.exe -sL $tamUrl) -join [Environment]::NewLine } if ([string]::IsNullOrWhiteSpace($kod) -or $kod -match "DOCTYPE" -or $kod -match "pages.github") { try { [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12 $kod = Invoke-RestMethod -Uri $tamUrl -ErrorAction Stop } catch {} } if ([string]::IsNullOrWhiteSpace($kod) -or $kod -match "DOCTYPE" -or $kod -match "pages.github") { Write-Host "HATA: Arac sunucudan cekilemedi!" -ForegroundColor Red [Console]::ReadLine() | Out-Null return } try { $script:altAracAktif = $true & ([scriptblock]::Create($kod)) return } catch { if ($_.Exception.Message -eq "ALT_MENU_DON") { continue } if ($_.Exception.Message -eq "ANA_MENU_DON") { return } Write-Host "HATA: $($_.Exception.Message)" -ForegroundColor Red [Console]::ReadLine() | Out-Null return } finally { $script:altAracAktif = $false } } } while ($true) { Clear-Host Write-Host "=== FRKOMER SYSTEM & SETUP CONTROL CENTER ===" -ForegroundColor Cyan Write-Host "1 - Office Setup & Management Tool (Office 2021/2024 Indir, Kur, Etkinlestir)" -ForegroundColor Yellow Write-Host "2 - Windows ISO & Upgrade Tool (Windows 10/11 ISO Indir, Yukselt, Format USB)" -ForegroundColor Yellow Write-Host "3 - Automatic Software Installer (Chrome, WinRAR, Adobe, AnyDesk vb. Kur)" -ForegroundColor Yellow Write-Host "4 - System Maintenance & Drivers (Temizlik, Onarim, Surucu, Imaj ve Klonlama)" -ForegroundColor Yellow Write-Host "5 - System Diagnostics & License (Donanim, BIOS Lisans, Disk/Pil Sagligi, Ag)" -ForegroundColor Yellow Write-Host "0 - Exit (Cikis)" -ForegroundColor Gray Write-Host "99 - Refresh Main Menu (Ana Menuyu Yenile / Basa Don)" -ForegroundColor DarkGray Write-Host "(Bilgi: Alt araclarda 0 yazarak o aracin basina, 99 yazarak Ana Menuye donebilirsiniz)" -ForegroundColor DarkGray Write-Host "" $secim = "" try { $secim = Read-Host "Select Tool / Arac Secin (0-5 veya 99)" } catch { continue } if ($secim -eq "0") { break } elseif ($secim -eq "99") { continue } elseif ($secim -eq "1") { Araci-Calistir "Office" } elseif ($secim -eq "2") { Araci-Calistir "Windows" } elseif ($secim -eq "3") { Araci-Calistir "Program" } elseif ($secim -eq "4") { Araci-Calistir "Bakim" } elseif ($secim -eq "5") { try { Sistem-Teshis-Menusu } catch {} } else { Write-Host "Gecersiz secim yaptiniz." -ForegroundColor Red Start-Sleep -Seconds 2 } }
